# Pinned for reproducible cloud deploys. Lower bound = the version this app was
# verified against; upper bound = next major, to block a surprise breaking
# release on a future rebuild. Bump deliberately, not by accident.
#
# yfinance is the most fragile dep: Yahoo periodically changes its undocumented
# endpoints and yfinance breaks until patched. If live data stops loading,
# bump yfinance FIRST (`pip install -U yfinance`) and re-pin here.

# Core numerical stack
numpy>=2.4,<3
pandas>=3.0,<4
scipy>=1.17,<2

# Visualisation
matplotlib>=3.7,<4   # notebook-only (core.simulator.plot); lazy-imported, never loaded by the app
plotly>=6.8,<7

# Market data
yfinance>=1.4,<2

# Optional: machine-translate pulled Yahoo descriptions into the UI language
# (the issuer / underlying "prefill from Yahoo" buttons). Best-effort — the app
# keeps the original English text if this is missing or the network is down.
deep-translator>=1.11,<2

# PDF report. The themed-document engine now lives in its own repo — this app is
# the Structured-Note ADAPTER over it (app/pdf_report.py). Pinned to a TAG, not a
# branch: an unpinned dependency would let a library change alter every client
# report on the next rebuild, silently.
#   https://github.com/diegogomezpy/report_maker
# `[charts]` pulls plotly + kaleido, which this app needs because it hands over
# Plotly figures rather than PNG bytes. fpdf2 and Pillow arrive as reportkit's
# own dependencies; they stay listed because app/ and api/ import them directly.
reportkit[charts] @ git+https://github.com/diegogomezpy/report_maker@v0.6.0
fpdf2>=2.8,<3
kaleido>=1.3,<2
Pillow>=12,<13

# Optional: shared run store for a multi-instance deploy. Only used when REDIS_URL
# is set (else the app uses the in-memory store); installed so setting REDIS_URL
# "just works" without an image rebuild. See api/engine.py:_make_run_store.
redis>=5,<7

# PyMuPDF (fitz) is NOT a runtime dep — it's only used by scripts/verify_pdf.py
# to rasterise PDFs for eyeballing. Install it ad hoc for verification:
#   pip install "PyMuPDF>=1.27,<2"

# PDF rasteriser for the PDF Studio's live proof (api/proof.py) — turns rendered
# report pages into PNGs. BSD/Apache-licensed with manylinux wheels, unlike
# PyMuPDF above (AGPL), which is why the proof path does not reuse it.
pypdfium2>=5,<6
